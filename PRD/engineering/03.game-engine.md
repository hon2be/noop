## GameWindow 클래스

```python
import curses, time

class GameWindow:
    TOGGLE_KEY = 28   # Ctrl+\
    QUIT_KEY   = 17   # Ctrl+Q

    def __init__(self, stdscr):
        self.stdscr = stdscr
        self.collapsed = False
        self.score = 0
        self.total = 10
        self.current = 3
        self.game_type = "블럭이동"

    def render(self):
        self.stdscr.clear()
        if self.collapsed:
            self._render_collapsed()
        else:
            self._render_expanded()
        self.stdscr.refresh()

    def _render_collapsed(self):
        rows, cols = self.stdscr.getmaxyx()
        bar = (
            f" ▶  {self.game_type}"
            f"  {self.current}/{self.total}"
            f"  │  Ctrl+\\ 펼치기 "
        ).ljust(cols)[:cols - 1]
        self.stdscr.attron(curses.A_REVERSE)
        self.stdscr.addstr(rows - 1, 0, bar)
        self.stdscr.attroff(curses.A_REVERSE)

    def _render_expanded(self):
        rows, cols = self.stdscr.getmaxyx()
        border = "─" * (cols - 2)
        self.stdscr.addstr(0, 0, f"┌{border}┐")
        self.stdscr.addstr(rows - 1, 0, f"└{border}┘")
        for y in range(1, rows - 1):
            self.stdscr.addstr(y, 0, "│")
            self.stdscr.addstr(y, cols - 1, "│")
        self.stdscr.addstr(1, 2,
            f" Q{self.current}/{self.total}  [{self.game_type}] ",
            curses.A_BOLD)
        self._render_game_content()
        self.stdscr.addstr(rows - 2, 2,
            " ←→ 이동  Space 선택  Enter 제출  Ctrl+\\ 접기 ",
            curses.A_DIM)

    def _render_game_content(self):
        pass  # subtype 렌더러 override

    def run(self):
        self.stdscr.nodelay(True)
        curses.curs_set(0)
        while True:
            self.render()
            key = self.stdscr.getch()
            if key == self.QUIT_KEY:
                break
            elif key == self.TOGGLE_KEY:
                self.collapsed = not self.collapsed
            elif not self.collapsed:
                self._handle_game_key(key)
            time.sleep(0.016)

    def _handle_game_key(self, key):
        pass  # subtype에서 override
```
